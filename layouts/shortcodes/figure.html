{{- /* Numbered figure shortcode with LaTeX-style cross-reference metadata. The actual number is assigned in document order by assets/js/custom.js. Keeping numbering out of the build template makes forward references work and prevents unrelated shortcodes from affecting the sequence. */ -}} {{- $src := cond .IsNamedParams (.Get "src") (.Get 0) -}} {{- $caption := "" -}} {{- $captionHTML := "" -}} {{- $alt := "" -}} {{- $label := "" -}} {{- $class := "" -}} {{- $link := "" -}} {{- $target := "" -}} {{- $rel := "" -}} {{- $width := "" -}} {{- $height := "" -}} {{- $loading := "lazy" -}} {{- $attr := "" -}} {{- $attrlink := "" -}} {{- $prefix := .Page.Params.figurePrefix | default site.Params.page.figurePrefix | default "Figure" -}} {{- $numbered := true -}} {{- if .IsNamedParams -}} {{- $caption = .Get "caption" | default (.Get "title") -}} {{- $captionHTML = $caption | .Page.RenderString (dict "display" "inline") -}} {{- $alt = .Get "alt" | default ($captionHTML | plainify) -}} {{- $label = .Get "label" | default "" -}} {{- $class = .Get "class" | default "" -}} {{- $link = .Get "link" | default "" -}} {{- $target = .Get "target" | default "" -}} {{- $rel = .Get "rel" | default "" -}} {{- $width = .Get "width" | default "" -}} {{- $height = .Get "height" | default "" -}} {{- $loading = .Get "loading" | default "lazy" -}} {{- $attr = .Get "attr" | default "" -}} {{- $attrlink = .Get "attrlink" | default "" -}} {{- $prefix = .Get "prefix" | default $prefix -}} {{- if isset .Params "numbered" -}} {{- $numbered = .Get "numbered" -}} {{- end -}} {{- end -}} {{- $figureID := "" -}} {{- with $label -}} {{- $labelName := strings.TrimPrefix "fig:" . -}} {{- $figureID = printf "fig-%s" ($labelName | anchorize) -}} {{- end -}} {{- if and .IsNamedParams (.Get "id") -}} {{- $figureID = .Get "id" -}} {{- end -}} {{- $classes := slice "article-figure" -}} {{- if $numbered -}} {{- $classes = $classes | append "numbered-figure" -}} {{- else -}} {{- $classes = $classes | append "unnumbered-figure" -}} {{- end -}} {{- with $class -}} {{- $classes = $classes | append . -}} {{- end -}}
